(defun oo-dwim-add-abbrev ()
  "Add an abbrev entry interactively and update the abbrev file."
  (interactive)
  (let* ((prev-word (save-excursion
                      (backward-word)
                      (thing-at-point 'word t)))
         (expansion (read-string (format "Expand '%s' to: " prev-word)))
         (abbrev-file "~/.emacs.d/abbrev_defs"))  ;; Adjust path if needed
    (when (and prev-word expansion (not (string-empty-p expansion)))
      (with-temp-buffer
        (insert-file-contents abbrev-file)
        (goto-char (point-max))
        (unless (bolp) (insert "\n")) ;; Ensure newline before appending
        (insert (format "%-20s \"%s\"\n" prev-word expansion)) ;; Align
        (write-region (point-min) (point-max) abbrev-file))
      (read-abbrev-file abbrev-file)
      (define-abbrev global-abbrev-table prev-word expansion)
      (message "Added abbrev: %s -> %s" prev-word expansion))))

#+begin_src emacs-lisp
(defun! oo-dwim-add-abbrev ()
  "Add an abbrev entry interactively and update the abbrev file."
  (interactive)
  (set! prev-word (save-excursion (backward-word) (thing-at-point 'word t)))
  (set! expansion (read-string (format "Expand '%s' to: " prev-word)))
  (set! lisp-dir (locate-user-emacs-file "lisp/"))
  ;; TODO ensure abbrev is not already defined.
  (quitif! () "Abbrev %S already expands to %S, aborting.")
  ;; TODO warn and ask if abbrev already expands to something else.
  (set! abbrev-file (expand-file-name "990-plain-text-abbrevs.el" "lisp/"))
  (quitif! (not prev-word) "No previous word found.")
  (quitif! (string-empty-p expansion) "No expansion given.")
  (set! form `(define-abbrev global-abbrev-table ,abbrev ,expansion nil :enable-function ))
  (with-temp-buffer
    (insert-file-contents abbrev-file)
    (goto-char (point-max))
    (unless (bolp) (insert "\n"))
    (insert (format "%s" form))
    (write-region (point-min) (point-max) abbrev-file))
  ;; (define-abbrev global-abbrev-table prev-word expansion)
  (message "Added abbrev: %s -> %s" prev-word expansion))
