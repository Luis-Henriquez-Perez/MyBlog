#+begin_src emacs-lisp
(defun! oo-dwim-add-abbrev ()
  "Add an abbrev entry interactively and update the abbrev file."
  (interactive)
  (set! abbrev (save-excursion (backward-word) (thing-at-point 'word t)))
  (set! expansion (read-string (format "Expand '%s' to: " abbrev)))
  (set! lisp-dir (locate-user-emacs-file "lisp/"))
  ;; TODO ensure abbrev is not already defined.
  ;; (quitif! () "Abbrev %S already expands to %S, aborting.")
  ;; TODO warn and ask if abbrev already expands to something else.
  (set! abbrev-file (expand-file-name "999-abbrevs.el" lisp-dir))
  (quitif! (not abbrev) "No previous word found.")
  (quitif! (string-empty-p expansion) "No expansion given.")
  ;; Find where the first `define-abbrev' starts.
  (set! form `(define-abbrev global-abbrev-table ,abbrev ,expansion nil :enable-function #'oo-use-text-abbrev-p))
  (with-current-buffer (get-buffer-create abbrev-file)
    (if (buffer-modified-p) (write-region (point-min) (point-max) abbrev-file))
    (goto-char (point-min))
    (re-search-forward "^(define-abbrev" nil t nil)
    (insert (format "%S" form))
    ;; (save-buffer)
    )
  (message "Added abbrev: %s -> %s" prev-word expansion))
