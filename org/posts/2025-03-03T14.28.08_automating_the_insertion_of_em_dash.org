#+begin_src emacs-lisp
(define-abbrev-table 'punctuation-abbrev-table
  '(("--" "—"))  ;; You can add more abbreviations here
  )
'(("--" "⸺"))  ;; You can add more abbreviations here
(alet! (rx (seq (1+ (not punct)) (group (one-or-more punct)) (zero-or-more (not punct))))
  (abbrev-table-put punctuation-abbrev-table :regexp it))

(setq-local local-abbrev-table (cons punctuation-abbrev-table (ensure-list local-abbrev-table)))
