;; Defined in ~/.config/emacs/elpa/emms-20250204.2022/emms.el
(defun emms-random ()
  "Jump to a random track."
  (interactive)
  (when emms-player-playing-p
    (emms-stop))
  (emms-playlist-current-select-random)
  (emms-start))
