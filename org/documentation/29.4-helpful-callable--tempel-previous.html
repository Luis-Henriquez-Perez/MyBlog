;; Defined in ~/.config/emacs/elpa/tempel-20250101.927/tempel.el
(defun tempel-previous (arg)
  "Move ARG fields backward and quit at the beginning."
  (declare (completion tempel--active-p))
  (interactive "p")
  (tempel-next (- arg)))
