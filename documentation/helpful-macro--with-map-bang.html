;; Defined in ~/.config/emacs/lisp/oo-base-macros-with-map-bang.el
(defmacro with-map! (map &rest body)
  "Let-bind bang symbols in BODY to corresponding keys in MAP."
  (declare (indent 1))
  `(let* ,(oo--map-let-binds map body "!\\([^[:space:]]+\\)" nil)
     ,@body))
