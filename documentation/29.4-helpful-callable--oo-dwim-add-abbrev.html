;; Closure converted to defun by helpful.
(defun oo-dwim-add-abbrev nil "Add an abbrev entry interactively and update the abbrev file."
       (interactive)
       (let
           ((form nil)
            (abbrev-file nil)
            (lisp-dir nil)
            (expansion nil)
            (abbrev nil))
         (catch 'return!
           (setq abbrev
                 (downcase
                  (save-excursion
                    (backward-word)
                    (thing-at-point 'word t))))
           (setq expansion
                 (read-string
                  (format "Expand '%s' to: " abbrev)))
           (setq lisp-dir
                 (locate-user-emacs-file "lisp/"))
           (setq abbrev-file
                 (expand-file-name "999-abbrevs.el" lisp-dir))
           (if
               (not abbrev)
               (progn
                 (throw 'return! nil)
                 (message "No previous word found.")))
           (if
               (string-empty-p expansion)
               (progn
                 (throw 'return! nil)
                 (message "No expansion given.")))
           (setq form
                 (cons 'define-abbrev
                       (cons 'global-abbrev-table
                             (cons abbrev
                                   (cons expansion
                                         '(nil :enable-function #'oo-use-text-abbrev-p))))))
           (save-current-buffer
             (set-buffer
              (or
               (get-file-buffer abbrev-file)
               (create-file-buffer abbrev-file)))
             (if
                 (buffer-modified-p)
                 (write-region
                  (point-min)
                  (point-max)
                  abbrev-file))
             (goto-char
              (point-min))
             (or
              (re-search-forward "^(define-abbrev" nil t nil)
              (re-search-forward "^;;; provide" nil t nil)
              (error "Could not find place to add abbrev."))
             (goto-char
              (match-beginning 0))
             (insert
              (format "%S\n" form)))
           (eval form 'lexical)
           (expand-abbrev)
           (message "Added abbrev: %s -> %s" abbrev expansion))))
